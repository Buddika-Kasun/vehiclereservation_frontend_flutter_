# Simpler optimized version
FROM debian:bullseye-slim AS build

RUN apt-get update && apt-get install -y curl git unzip
RUN git clone https://github.com/flutter/flutter.git --depth 1 -b stable
ENV PATH="$PATH:/flutter/bin"

WORKDIR /app
COPY . .

RUN mkdir -p assets && touch assets/.env
RUN flutter pub get

ENV FLUTTER_WEB_USE_SKIA=false
RUN flutter build web --release --no-source-maps

# ====== FORCE UPDATE FOR ALL USERS NOW ======
# Generate unique build ID
RUN BUILD_ID=$(date +%s) && \
    # 1. Add version to ALL assets
    sed -i "s/main\.dart\.js/main.dart.js?v=$BUILD_ID/g" build/web/index.html && \
    sed -i "s/flutter\.js/flutter.js?v=$BUILD_ID/g" build/web/index.html && \
    sed -i "s/main\.css/main.css?v=$BUILD_ID/g" build/web/index.html && \
    # 2. ADD SERVICE WORKER REMOVAL SCRIPT
    sed -i '/<head>/a\
    <script>\
    // FORCE UPDATE FOR EXISTING USERS\
    if ("serviceWorker" in navigator) {\
      window.addEventListener("load", function() {\
        navigator.serviceWorker.getRegistrations().then(function(registrations) {\
          for (let registration of registrations) {\
            registration.unregister();\
            console.log("ServiceWorker unregistered");\
          }\
        });\
        \
        // Clear ALL caches\
        if ("caches" in window) {\
          caches.keys().then(function(keyList) {\
            return Promise.all(keyList.map(function(key) {\
              return caches.delete(key);\
            }));\
          });\
        }\
      });\
    }\
    </script>' build/web/index.html && \
    # 3. Delete service worker file
    rm -f build/web/flutter_service_worker.js 2>/dev/null || true && \
    rm -f build/web/service-worker.js 2>/dev/null || true

# Create config.js
RUN echo 'window.config = {' > build/web/config.js
RUN echo '  apiUrl: "${API_URL:-https://api.example.com}",' >> build/web/config.js
RUN echo '  wsUrl: "${WS_URL:-wss://ws.example.com}",' >> build/web/config.js
RUN echo '};' >> build/web/config.js

FROM nginx:alpine
COPY --from=build /app/build/web /usr/share/nginx/html

# ====== TEMPORARY: DISABLE ALL CACHING ======
# Write nginx config in one RUN command to avoid parsing issues
RUN echo "events{}\
http {\
  server {\
    listen 8080;\
    root /usr/share/nginx/html;\
    \
    # TEMPORARY: NO CACHING AT ALL\
    add_header Cache-Control 'no-store, no-cache, must-revalidate, proxy-revalidate, max-age=0';\
    add_header Pragma 'no-cache';\
    add_header Expires '0';\
    \
    location / {\
      try_files \$uri \$uri/ /index.html;\
    }\
  }\
}" > /etc/nginx/nginx.conf

EXPOSE 8080
CMD ["nginx", "-g", "daemon off;"]
